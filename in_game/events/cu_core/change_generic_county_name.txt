namespace = cu_change_generic_county_name_event

cu_change_generic_county_name_event.1 = {
	type = country_event
	hidden = yes
	triggered_only = yes

	trigger = {
		country_rank = country_rank:rank_county
		NOT = { culture = culture:sumerian_culture }
	}

	immediate = {
		if = { limit = { capital = location:abadan_location } change_country_name = abadan_location }
		else_if = { limit = { capital = location:nasiriyah_location } change_country_name = nasiriyah_location }
        else_if = { limit = { capital = location:suq_al_shuyukh_location } change_country_name = suq_al_shuyukh_location }
	}

	option = {
		name = cu_change_generic_county_name_event.1.ok
	}
}
