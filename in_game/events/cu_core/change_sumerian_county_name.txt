namespace = cu_change_sumerian_county_name_event

cu_change_sumerian_county_name_event.1 = {
	type = country_event
	hidden = yes
	triggered_only = yes

	trigger = {
		country_rank = country_rank:rank_county
  		culture = culture:sumerian_culture
	}

	immediate = {
		if = { limit = { capital = location:abadan_location } change_country_name = abadan_location.sumerian_language }
		else_if = { limit = { capital = location:nasiriyah_location } change_country_name = nasiriyah_location.sumerian_language }
        else_if = { limit = { capital = location:samawah_location } change_country_name = samawah_location.sumerian_language }
        else_if = { limit = { capital = location:shatrah_location } change_country_name = shatrah_location.sumerian_language }
        else_if = { limit = { capital = location:suq_al_shuyukh_location } change_country_name = suq_al_shuyukh_location.sumerian_language }
        else_if = { limit = { capital = location:tell_abu_shahrain_location } change_country_name = tell_abu_shahrain_location.sumerian_language }
        else_if = { limit = { capital = location:tell_al_lahm_location } change_country_name = tell_al_lahm_location.sumerian_language }
	}

	option = {
		name = cu_change_sumerian_county_name_event.1.ok
	}
}
